26class ProcessStackHandler {
27 public:
28 explicit ProcessStackHandler(InternalInterface *internal) {
29 // The process stack is used to guard re-entry to make sure we exit
30 // SystemC to Simics on the same thread we entered on. But we only need
31 // to grow the stack if we are leaving on a thread, since METHOD and
32 // NO_PROC will trigger an error on re-entry if used for normal
33 // b_transport. See simics/systemc/simics2tlm/gasket.cc for details.
34 if (sc_core::sc_get_status() == sc_core::SC_RUNNING) {
35 sc_core::sc_curr_proc_kind kind =
36 sc_core::sc_get_current_process_handle().proc_kind();
37 if (kind == sc_core::SC_THREAD_PROC_
38 || kind == sc_core::SC_CTHREAD_PROC_) {
39 pstack_ = internal->process_stack();
40 pstack_->push();
41 }
42 }
43 }
44 ProcessStackHandler(const ProcessStackHandler&) = delete;
45 ProcessStackHandler& operator=(const ProcessStackHandler&) = delete;
46 virtual ~ProcessStackHandler() {
47 if (pstack_) {
48 pstack_->pop();
49 }
50 }
51
52 private:
53 ProcessStackInterface *pstack_ {nullptr};
54};
