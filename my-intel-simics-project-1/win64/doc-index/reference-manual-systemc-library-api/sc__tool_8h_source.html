133template <class T>
134simics::ConfClassPtr ScTool::register_class(std::string name,
135 std::string long_name,
136 std::string documentation) {
137 auto cls = simics::make_class<T>(name, long_name,
138 documentation,
139 Sim_Class_Kind_Pseudo);
140 cls->add(scl::iface::createAdapter<scli::ToolSimicsAdapter<T>>());
141 return cls;
142}
