37template<typename TExtension, typename TInterface>
38class Extension : public tlm::tlm_extension<TExtension>, public TInterface {
39 public:
40 Extension() : sender_(NULL), method_call_count_(0),
41 suppress_error_(false), valid_(false) {}
43 void init(ExtensionSenderInterface *sender,
44 tlm::tlm_generic_payload *payload) {
45 sender_ = sender;
46 }
47 void init(ExtensionSenderInterface *sender) {
48 sender_ = sender;
49 }
50 void send() {
51 // Clear returns from former call.
52 method_return_ = types::AnyType();
53 method_call_count_ = 0;
54 suppress_error_ = false;
55 // Copy method, method_input and method_return_error into new
56 // extension object.
57 Extension *e = static_cast<Extension*>(clone());
58 // Clear method, method_input and method_return_error for reenter
59 // from send_extension.
60 method_input_.clear();
61 method_ = types::AnyType();
62 method_return_error_ = types::AnyType();
63 e->valid_ = true;
64 Transaction t = sender_->transaction();
65 t->set_extension(e);
66 /* During send_extension, the extension is dispatched to an
67 * ExtensionReceiver or an ExtensionIgnoreReceiver. The receiver
68 * invokes either method_call or method_call_ignore.*/
69 sender_->send_extension(&t);
70 // Copy method_return back.
71 if (e->method_call_count_ != 1) {
72 method_return_ = e->method_return_error_;
73 if (!method_return_.isSet())
74 method_return_ = 0;
75 if (!e->suppress_error_)
76 sender_->send_failed(&t);
77 } else {
78 method_return_ = e->method_return_;
79 }
80 t->clear_extension(e);
81 e->free();
82 }
83 void method_call(TInterface *device) {
84 ++method_call_count_;
85 if (device)
86 call(device);
87 }
88 void method_call_ignore() {
89 suppress_error_ = true;
90 }
91 types::AnyType method_type() {
92 return method_;
93 }
94 bool valid() {
95 return valid_;
96 }
97 tlm::tlm_extension_base *clone() const override {
98 return new TExtension(*static_cast<const TExtension *>(this));
99 }
100 void copy_from(tlm::tlm_extension_base const &extension) override {
101 *this = static_cast<const TExtension &>(extension);
102 }
103 static ReceiverInterface *createReceiver(TInterface *device) {
104 return new ExtensionReceiver<TExtension, TInterface>(device);
105 }
106 static ReceiverInterface *createIgnoreReceiver() {
107 return new ExtensionIgnoreReceiver<TExtension>();
108 }
109 virtual ~Extension() {}
110
111 protected:
112 virtual void call(TInterface *device) = 0;
113
114 ExtensionSenderInterface *sender_;

116 std::vector<types::AnyType> method_input_;
117 types::AnyType method_return_;
118 types::AnyType method_return_error_;
119 int method_call_count_;
120 bool suppress_error_;
121 bool valid_;
122};
