29class SpiSlaveGasketAdapter
30 : public iface::SpiSlaveInterface,
31 public GasketAdapter<iface::SpiSlaveInterface> {
32 public:
33 SpiSlaveGasketAdapter(SpiSlaveInterface *spi_slave,
34 iface::SimulationInterface *simulation,
35 ConnectBase *spi_master)
36 : spi_slave_(spi_slave), simulation_(simulation),
37 spi_master_(spi_master) {
38 assert(spi_master_);
39 }
40 // iface::SpiSlaveInterface
41 void spi_request(int first, int last, const uint8 *data_ptr,
42 size_t data_length) override {
43 Context context(simulation_);
44 spi_slave_->spi_request(first, last, data_ptr, data_length);
45 }
46 void connect_master(conf_object_t *master, const char *port,
47 serial_peripheral_interface_flags_t flags) override {
48 auto *prev_master = spi_master_->get().object();
49 if (prev_master) {
50 SIM_LOG_INFO(3, simulation_->simics_object(), 0,
51 "disconnected from master %s",
52 SIM_object_name(prev_master));
53 }
54
55 ConfObjectRef obj {master};
56 if (port) {
57 obj.set_port_name(port);
58 }
59 spi_master_->set(obj);
60 SIM_LOG_INFO(3, simulation_->simics_object(), 0,
61 "connected to master %s", obj.name().c_str());
62
63 flags_ = flags;
64 }
65 void disconnect_master(conf_object_t *master) override {
66 if (spi_master_->get().object() != master) {
67 SIM_LOG_ERROR(simulation_->simics_object(), 0,
68 "can not be disconnected from master device %s",
69 SIM_object_name(master));
70 } else {
71 spi_master_->set(nullptr);
72 flags_ = -1;
73 SIM_LOG_INFO(3, simulation_->simics_object(), 0,
74 "disconnected from master %s",
75 SIM_object_name(master));
76 }
77 }
78 simics2tlm::GasketOwner *gasket_owner() const override {
79 return dynamic_cast<simics2tlm::GasketOwner *>(spi_slave_);
80 }
81
84 int flags() {
85 if (spi_master_->get().object() == nullptr) {
86 return -1;
87 }
88 return flags_;
89 }
90
91 private:
92 SpiSlaveInterface *spi_slave_;
93 iface::SimulationInterface *simulation_;
94 ConnectBase *spi_master_;
95 int flags_ {-1};
96};
