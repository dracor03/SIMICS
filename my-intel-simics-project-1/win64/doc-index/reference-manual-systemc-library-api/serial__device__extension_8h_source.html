25class SerialDeviceExtension : public Extension<SerialDeviceExtension,
26 SerialDeviceInterface> {
27 public:
28 void call(SerialDeviceInterface *device) override {
29 switch (method_.value<Method>()) {
30 case WRITE:
31 method_return_ = device->write(method_input_[0].value<int>());
32 break;
33 case RECEIVE_READY:
34 device->receive_ready();
35 break;
36 }
37 }
38
39 int write(int value) override {
40 method_input_.push_back(value);
41 method_ = WRITE;
42 send();
43 return method_return_.value<int>();
44 }
45 void receive_ready() override {
46 method_ = RECEIVE_READY;
47 send();
48 }
49
50 private:
51 enum Method {
52 WRITE,
53 RECEIVE_READY
54 };
55};
