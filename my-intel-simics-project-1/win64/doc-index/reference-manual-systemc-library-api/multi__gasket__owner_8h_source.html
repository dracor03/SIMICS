38class MultiGasketOwner : public GasketOwner {
39 public:
40 virtual ~MultiGasketOwner() = default;
41
42 void addGasket(int id, GasketInterface::Ptr gasketInterface) {
43 FATAL_ERROR_IF(hasId(id),
44 "Cannot bind the same ID (%d) to multiple gaskets", id);
45 gasket_map_.emplace(std::piecewise_construct,
46 std::forward_as_tuple(id),
47 std::forward_as_tuple());
48 // Workaround: Set ClassType::implementor_ using this class's value
49 gasket_map_[id].ClassType::operator=(*static_cast<ClassType *>(this));
50 gasket_map_[id].set_gasket(std::move(gasketInterface));
51 // Need to delete NullGasket if it exists
52 if (dynamic_cast<NullGasket*>(gasket_.get()))
53 gasket_ = NULL;
54 }
55
63 GasketInterface::Ptr findGasket(int id) const {
64 GasketOwnerMap::const_iterator it = gasket_map_.find(id);
65 if (it != gasket_map_.end()) {
66 return it->second.gasket();
67 } else if (!empty()) {
68 return gasket_map_.begin()->second.gasket();
69 } else {
70 return gasket();
71 }
72 }
73 std::set<int> keys() const {
74 std::set<int> k;
75 for (auto i = gasket_map_.begin(); i != gasket_map_.end(); ++i)
76 k.insert(i->first);
77
78 return k;
79 }
80
81 protected:
82 bool hasId(int id) const {
83 return gasket_map_.count(id);
84 }
85 int empty() const {
86 return gasket_map_.empty();
87 }
88
89 private:
90 typedef std::map<int, GasketOwner> GasketOwnerMap;
91 GasketOwnerMap gasket_map_;
92};
