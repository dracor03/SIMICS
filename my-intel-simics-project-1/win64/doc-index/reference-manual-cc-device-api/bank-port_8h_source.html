77 /*
78 * CTOR used for the "by code" modeling option
79 * "by code" option is used when the bank is created and initialized
80 * explicitly by code as a class member variable of a bank port.
81 */
82 explicit BankPort(ConfObjectRef o): Port<TParent>(o) {
83 set_bank_name_from_port_name(o.name());
84 SIM_hap_add_callback_obj(

139 void set_bank(const bank_t &bank) override {
140 if (bank_iface_) {
141 SIM_LOG_ERROR(Port<TParent>::obj(), 0,
142 "bank iface can only be set once");
143 return;
144 }
145
146 bank_iface_ = dev_obj()->template get_iface<BankInterface>(bank_name_);
147 if (bank_iface_) {
148 SIM_LOG_INFO(3, Port<TParent>::obj(), 0,
149 "Used user defined bank for %s",
150 bank_name_.c_str());
151 } else {
152 allocated_bank_.reset(new Bank(dev_obj(), bank_name_));
153 bank_iface_ = allocated_bank_.get();
154 SIM_LOG_INFO(3, Port<TParent>::obj(), 0,
155 "Created a new default bank for %s",
156 bank_name_.c_str());
157 }
158 auto &[name, desc, registers] = bank;
159 bank_iface_->set_description(desc);
160 bank_iface_->set_callbacks(this);
161 for (const auto &reg : registers) {
162 bank_iface_->add_register(reg);
163 }
164 }

219 // iface::RegisterViewCatalogInterface
220 attr_value_t register_names() override {
221 auto number_of_regs = number_of_registers();
222 attr_value_t ret = SIM_alloc_attr_list(number_of_regs);
223 RegisterInterface *iface = nullptr;
224 for (unsigned i = 0; i < number_of_regs; ++i) {
225 std::tie(std::ignore, iface) = register_at_index(i);
226 if (iface == nullptr) {
227 SIM_LOG_ERROR(Port<TParent>::obj(), 0,
228 "Invalid register index %d", i);
229 SIM_attr_list_set_item(&ret, i, SIM_make_attr_nil());
230 continue;
231 }
232 SIM_attr_list_set_item(&ret, i,
233 SIM_make_attr_string(iface->name().data()));
234 }
235 return ret;
236 }

256 // Initialize the register offsets for better performance
257 // when using the register_view interface
258 void init_register_offsets() {
259 const auto &mapped_regs = bank_iface_->mapped_registers();
260 regs_offsets_.clear();
261 regs_offsets_.reserve(mapped_regs.size());
262 for (const auto& [offset, reg_iface] : mapped_regs) {
263 regs_offsets_.push_back(offset);
264 }
265 }

268 static void object_created(lang_void *, conf_object_t *obj) {
269 auto *iface = detail::get_interface<BankPort<TParent>>(obj);
270 if (iface->validate_bank_iface()) {
271 // Register can't be added after the bank port is finalized,
272 // so it is safe to initialize the register offsets now
273 iface->init_register_offsets();
274 }

277 static attr_value_t register_info(size_t address, ByteOrder bo,
278 const RegisterInterface *i) {
279 attr_value_t info = SIM_alloc_attr_list(6);
280 SIM_attr_list_set_item(&info, 0,
281 SIM_make_attr_string(i->name().data()));
282 SIM_attr_list_set_item(&info, 1,
283 SIM_make_attr_string(i->description().c_str()));
284 SIM_attr_list_set_item(&info, 2,
285 SIM_make_attr_uint64(i->number_of_bytes()));
286 SIM_attr_list_set_item(&info, 3, SIM_make_attr_uint64(address));
287
288 auto fields_info = i->fields_info();
289 // The forth attr in info is a list fields
290 attr_value_t fields = SIM_alloc_attr_list(
291 checked_unsigned_cast(fields_info.size()));
292 unsigned index = 0;
293 for (const auto &[name, desc, offset, width] : fields_info) {
294 // Each attr in fields is another list field_info
295 attr_value_t field_info = SIM_alloc_attr_list(4);
296 SIM_attr_list_set_item(&field_info, 0,
297 SIM_make_attr_string(name.data()));
298 SIM_attr_list_set_item(&field_info, 1,
299 SIM_make_attr_string(desc.data()));
300 SIM_attr_list_set_item(&field_info, 2,
301 SIM_make_attr_uint64(offset));
302 SIM_attr_list_set_item(&field_info, 3,
303 SIM_make_attr_uint64(offset + width - 1));
304 SIM_attr_list_set_item(&fields, index++, field_info);
305 }
306 SIM_attr_list_set_item(&info, 4, fields);
307 SIM_attr_list_set_item(&info, 5,
308 SIM_make_attr_boolean(bo == ByteOrder::BE));
309 return info;
310 }

314 void set_bank_name_from_port_name(const std::string &port_name) {
315 auto pos = port_name.rfind(".bank.");
316 if (pos == std::string::npos) {
317 throw std::invalid_argument(
318 "Invalid bank port name (" + port_name + ")");
319 }
320 bank_name_ = port_name.substr(pos + 6);
321 }
322
323 std::pair<size_t, RegisterInterface *> register_at_index(unsigned index) {
324 size_t reg_offset {0};
325 RegisterInterface *reg_iface {nullptr};
326 try {
327 reg_offset = regs_offsets_.at(index);
328 } catch (const std::out_of_range &) {
329 SIM_LOG_ERROR(Port<TParent>::obj(), 0,
330 "Invalid register index %d", index);
331 return {reg_offset, reg_iface};
332 }
333 if (validate_bank_iface()) {
334 reg_iface = bank_iface_->mapped_registers().at(reg_offset);
335 }
336 return {reg_offset, reg_iface};
337 }
338
339 // Points to the actual bank interface
340 BankInterface *bank_iface_ { nullptr };
341 // Name of the bank
342 std::string bank_name_;
343 // @internal: record the heap allocated Bank
344 std::unique_ptr<BankInterface> allocated_bank_;
345 // @internal: list of all register offsets by ascending order
346 std::vector<size_t> regs_offsets_;
347};

433template <typename TParent>
434void create_hierarchy_from_register_data(ConfClass *cls, const bank_t &bank) {
435 const auto &[name, desc, registers] = bank;
436 cls->add(make_bank_port<BankPort<TParent>, const bank_t *>(
437 cls->name() + SEPARATOR + std::string(name.base_name()),
438 desc.data(), &bank),
439 std::string("bank.") + name.data());
440}
441// When passing initializer_list by value, only the wrapper (which contains
442// pointers to the elements) is copied, not the elements themselves.
443template <typename TParent>
444void create_hierarchy_from_register_data(
445 ConfClass *cls, std::initializer_list<bank_t> register_data) {
446 for (const auto &bank : register_data) {
447 create_hierarchy_from_register_data<TParent>(cls, bank);
448 }
