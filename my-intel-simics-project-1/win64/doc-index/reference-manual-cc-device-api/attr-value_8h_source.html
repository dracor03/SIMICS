23/*
24 * RAII class holding a Simics attr_value_t
25 *
26 * Construct the class from Simics API call which returns
27 * a attr_value_t, e.g, SIM_get_attribute, SIM_make_attr_...
28 * When desctructed, it frees the memory allocation used by
29 * the attr_value_t.
30 */
31class AttrValue {
32 public:
33 explicit AttrValue(attr_value_t &&attr) : attr_(std::move(attr)) {}
34
35 AttrValue(const AttrValue&) = delete;
36 AttrValue& operator=(const AttrValue &) = delete;
37 AttrValue(AttrValue &&rhs) noexcept : attr_(std::move(rhs.attr_)) {
38 rhs.attr_ = SIM_make_attr_invalid();
39 }
40 AttrValue &operator=(AttrValue &&rhs) noexcept {
41 if (this != &rhs) {
42 SIM_attr_free(&attr_);
43 attr_ = std::move(rhs.attr_);
44 rhs.attr_ = SIM_make_attr_invalid();
45 }
46 return *this;
47 }
48 ~AttrValue() {
49 SIM_attr_free(&attr_);
50 }
51
52 operator attr_value_t &() { return attr_; }
53
54 // Assignment operator from rvalue attr_value_t
55 AttrValue &operator=(attr_value_t &&rhs) noexcept {
56 SIM_attr_free(&attr_);
57 attr_ = std::move(rhs);
58 rhs = SIM_make_attr_invalid();
59 return *this;
60 }
61
62 private:
63 attr_value_t attr_;
64};
