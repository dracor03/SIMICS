34/*
35 * Register with map information
36 * This class creates a register object and add it to the <arg>bank_iface</arg>.
37 * Optional <arg>fields</arg> is used to create a field object and add it
38 * to the register.
39 * Customized registers such as ReadConstantRegister takes additional arguments.
40 */
41template <typename TRegister = Register, typename... Args>
42class BankRegister : public TRegister {
43 static_assert(std::is_base_of<Register, TRegister>::value,
44 "TRegister must be derived from Register");
45 public:
46 BankRegister(BankInterface *bank_iface,
47 Name reg_name, Description desc, Offset offset, ByteSize size,
48 InitValue value, std::initializer_list<field_t> fields = {},
49 Args... args)
50 : BankRegister(makeRegisterCtorParams(bank_iface, reg_name),
51 args...) {
52 bank_iface->add_register(reg_name, desc, offset, size, value, fields);
53 }
54
55 private:
57 BankRegister(std::pair<MappableConfObject*, std::string> &&params,
58 Args... args)
59 : TRegister(params.first, params.second, args...) {}
60};

126// Software can only clear bits. This feature is often used when
127// hardware sets bits and software clears them to acknowledge.
128// Software write 1's to clear bits. The new object value is a
129// bitwise AND of the old object value and the bitwise complement
130// of the value written by software.
131class Write1ClearsRegister : public Register {
132 public:
133 using Register::Register;
134
135 void write(uint64_t value, uint64_t enabled_bits) override {
136 Register::write(~value, enabled_bits & value);
137 }

439// The register is excluded from the address space of the containing bank.
440class UnmappedRegister : public Register {
441 public:
442 UnmappedRegister(MappableConfObject *obj, const std::string &name,
443 size_t number_of_bytes = 4, uint64_t init_value = 0)
444 : Register(obj, name) {
445 create_unmapped_register(number_of_bytes, init_value);
446 }
447
448 bool is_mapped() const override {
449 return false;
450 }
451
452 private:
453 void create_unmapped_register(size_t number_of_bytes,
454 uint64_t init_value) {
455 if (number_of_bytes > 8 || number_of_bytes == 0) {
456 SIM_LOG_ERROR(bank_obj_ref(), 0,
457 "The supported register size is [1-8] bytes");
458 return;
459 }
460 register_memory_t bytePointers(number_of_bytes);
461 std::iota(bytePointers.begin(), bytePointers.end(),
462 register_memory_.data());
463 set_byte_pointers(bytePointers);
464 init("Unmapped. ", number_of_bytes, init_value);
465 }
466
467 std::array<uint8_t, 8> register_memory_ {};
468};

486// The register is an alias for another register. All operations are forwarded
487// to the other register.
488class AliasRegister : public Register {
489 public:
490 AliasRegister(MappableConfObject *obj, const std::string &name,
491 const std::string &alias_name)
492 : Register(obj, name),
493 alias_name_(alias_name) {
494 init_alias();
495 }
496
497 bool is_read_only() const override {
498 return alias_->is_read_only();
499 }
500
501 bool is_mapped() const override {
502 return alias_->is_mapped();
503 }
504
505 uint64_t get() const override {
506 return alias_->get();
507 }
508
509 void set(uint64_t value) override {
510 alias_->set(value);
511 }
512
513 uint64_t read(uint64_t enabled_bits) override {
514 return alias_->read(enabled_bits);
515 }
516
517 void write(uint64_t value, uint64_t enabled_bits) override {
518 alias_->write(value, enabled_bits);
519 }
520
521 std::vector<field_t> fields_info() const override {
522 return alias_->fields_info();
523 }
524
525 private:
526 void init_alias() {
527 if (!HierarchicalObject::is_valid_hierarchical_name(alias_name_)
528 || HierarchicalObject::level_of_hierarchical_name(alias_name_) !=
529 static_cast<int>(
530 HierarchicalObjectInterface::Level::REGISTER)) {
531 std::string err = "Ignored invalid register name (" \
532 + alias_name_ + ")";
533 SIM_LOG_ERROR_STR(bank_obj_ref(), 0, err);
534 throw std::invalid_argument { err };
535 }
536 set_description("Alias register for register " + alias_name_ + ". "
537 + description());
538
539 alias_ = dev_obj()->get_iface<RegisterInterface>(alias_name_);
540 if (!alias_) {
541 // A limitation that depends on the register define order
542 std::string err = "The aliased register " + alias_name_ \
543 + " not found. Alter the register define order to make sure " \
544 + "it is defined before this register.";
545 SIM_LOG_ERROR_STR(bank_obj_ref(), 0, err);
546 throw std::invalid_argument { err };
547 }
548 }
549
550 std::string alias_name_;
551 RegisterInterface *alias_ {nullptr};
552};
