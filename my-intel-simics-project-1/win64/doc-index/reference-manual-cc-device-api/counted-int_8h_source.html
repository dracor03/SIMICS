21class CountedInt {
22 public:
23 explicit CountedInt(int val = 0) : value(val) {}
24
25 // Copy constructor
26 CountedInt(const CountedInt &other) : value(other.value) {
27 ++copyConstructorCalls;
28 std::cout << "Copy constructor called. Total calls: "
29 << copyConstructorCalls << std::endl;
30 }
31
32 // Copy assignment operator
33 CountedInt& operator=(const CountedInt &other) {
34 if (this != &other) {
35 value = other.value;
36 ++copyAssignmentCalls;
37 std::cout << "Copy assignment called. Total calls: "
38 << copyAssignmentCalls << std::endl;
39 }
40 return *this;
41 }
42
43 // Getter for the value
44 int getValue() const { return value; }
45
46 // Static methods to get the counts
47 static int getCopyConstructorCalls() { return copyConstructorCalls; }
48 static int getCopyAssignmentCalls() { return copyAssignmentCalls; }
49
50 // Reset counters (for testing purposes)
51 static void resetCounters() {
52 copyConstructorCalls = 0;
53 copyAssignmentCalls = 0;
54 }
55
56 private:
57 int value;
58 static inline int copyConstructorCalls {0};
59 static inline int copyAssignmentCalls {0};
60};
