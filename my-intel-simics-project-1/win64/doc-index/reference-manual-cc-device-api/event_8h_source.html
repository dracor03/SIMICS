117template <typename T = ConfObject>
118class TimeEvent : public Event,
119 public TimeEventInterface {
120 public:
121 using Event::Event;
122
123 // TimeEventInterface
124 bool posted(void *match_data = nullptr) const override {
125 return next(match_data) >= 0.0;
126 }
127
128 void remove(void *match_data = nullptr) const override {
129 if (clock_) {
130 SIM_event_cancel_time(clock_, ev_, obj_->obj(),
131 Event::pointer_eq, match_data);
132 }
133 }
134
135 void post(double seconds, void *data = nullptr) override {
136 if (clock_ == nullptr) {
138 clock_ = SIM_object_clock(obj_->obj());
139 if (clock_ == nullptr) {
140 SIM_LOG_ERROR(obj_->obj(), 0,
141 "Queue not set, unable to post events");
142 return;
143 }
144 }
145 SIM_event_post_time(clock_, ev_, obj_->obj(), seconds, data);
146 if (SIM_clear_exception() != SimExc_No_Exception) {
147 SIM_LOG_ERROR(obj_->obj(), 0, "%s", SIM_last_error());
148 }
149 }
150
151 double next(void *match_data = nullptr) const override {
152 if (clock_ == nullptr) {
153 return -1.0;
154 }
155 return SIM_event_find_next_time(clock_, ev_, obj_->obj(),
156 Event::pointer_eq, match_data);
157 }
158
159 protected:
160 T *dev_ {device_ptr<T>()};
161};
