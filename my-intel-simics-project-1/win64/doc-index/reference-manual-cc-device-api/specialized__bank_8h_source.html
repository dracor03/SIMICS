32 // Reset all registers in the bank

34 unsigned numOfRegs = number_of_registers();
35 for (unsigned i = 0; i < numOfRegs; ++i) {
36 std::pair<size_t, simics::RegisterInterface *> reg_pair = register_at_index(i);
37 reg_pair.second->reset();
38 }
