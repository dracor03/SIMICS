89 private:
90 class StrBuf : public std::stringbuf {
91 public:
92 explicit StrBuf(ConfObjectRef log_obj)
93 : std::stringbuf(ios_base::out),
94 log_obj_(log_obj.object()) {}
95
96 private:
97 int sync() override {
98 std::string msg = str();
99 // trim trailing newline (and whitespace)
100 auto pos = msg.find_last_not_of(" \t\n");
101 if (pos == std::string::npos) {
102 msg.clear();
103 } else if (pos != msg.length() - 1) {
104 msg.erase(pos + 1);
105 }
106
107 SIM_log_message(log_obj_, Level, Groups, Type, msg.c_str());
108 msg.clear();
109 str(msg);
110 return std::stringbuf::sync();
111 }
112
113 conf_object_t *log_obj_;
114 };
