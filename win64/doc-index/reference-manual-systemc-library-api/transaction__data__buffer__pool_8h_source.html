28/*
29 * The data buffer used for get/set the data from a transaction
30 */
31class TransactionDataBuffer {
32 public:
35 explicit TransactionDataBuffer(unsigned num_bytes)
36 : data_(num_bytes, 0) {
37 if (num_bytes == 0) {
38 throw std::invalid_argument {
39 "Cannot allocate an empty data buffer"
40 };
41 }
42 buffer_.data = &data_.front();
43 buffer_.len = num_bytes;
44 bytes_.data = &data_.front();
45 bytes_.len = num_bytes;
46 }
47
48 virtual ~TransactionDataBuffer() {
49 buffer_.data = nullptr;
50 bytes_.data = nullptr;
51 }
52
55 unsigned numBytes() const {
56 return data_.size();
57 }
58
60 unsigned char *dataPtr() const {
61 return buffer_.data;
62 }
63
64
66 void copyToBuffer(const transaction_t *transaction) {
67 auto size = SIM_transaction_size(transaction);
68 if (size > numBytes()) {
69 throw std::logic_error {
70 "The number of bytes for this data buffer ("
71 + std::to_string(numBytes())
72 + ") does not fit the size of the transaction ("
73 + std::to_string(size) + ")"
74 };
75 }
76 buffer_.len = size;
77 SIM_get_transaction_bytes(transaction, buffer_);
78 }
79
81 void copyFromBuffer(const transaction_t *transaction) {
82 auto size = SIM_transaction_size(transaction);
83 if (size > numBytes()) {
84 throw std::logic_error {
85 "The number of bytes for this data buffer ("
86 + std::to_string(numBytes())
87 + ") does not fit the size of the transaction ("
88 + std::to_string(size) + ")"
89 };
90 }
91 bytes_.len = size;
92 SIM_set_transaction_bytes(transaction, bytes_);
93 }
94
95 private:
96 std::vector<unsigned char> data_;
97 buffer_t buffer_;
98 bytes_t bytes_;
99};

101/*
102 * The pool for allocating/releasing TransactionDataBuffers
103 */
104class TransactionDataBufferPool {
105 public:
106 virtual ~TransactionDataBufferPool() {
107 for (auto it = pool_.begin(); it != pool_.end(); ++it) {
108 delete it->second;
109 }
110 }
111
112 TransactionDataBuffer *acquire(unsigned size) {
113 TransactionDataBuffer *buf = nullptr;
114 auto roundup_size = roundUpSize(size);
115 auto it = pool_.find(roundup_size);
116 if (it != pool_.end()) {
117 buf = it->second;
118 // Remove only one buffer of this size
119 pool_.erase(it);
120 } else {
121 buf = new TransactionDataBuffer(roundup_size);
122 }
123 return buf;
124 }
125
126 void release(TransactionDataBuffer *buf) {
127 pool_.emplace(buf->numBytes(), buf);
128 }
129
130 private:
131 // Round up requested size to the nearest size class:
132 // 8, 8K and 8M
133 static unsigned roundUpSize(unsigned size) {
134 if (size <= 8) {
135 return 8;
136 }
137 if (size <= 8192) {
138 return 8192;
139 }
140 if (size <= 8388608) {
141 return 8388608;
142 }
143 throw std::logic_error {
144 "The transaction data buffer does not support"
145 " data larger than 8M. Please report it."
146 };
147 }
148
149 std::unordered_multimap<unsigned, TransactionDataBuffer*> pool_;
150};
