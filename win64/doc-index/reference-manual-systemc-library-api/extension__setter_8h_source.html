31template <typename TPAYLOAD, typename TExtension>
32class ExtensionSetter : public iface::ExtensionSenderInterface {
33 public:
34 explicit ExtensionSetter(TPAYLOAD *gp)
35 : gp_at_call_(gp), extension_at_call_(NULL) {
36 extension_.init(this);
37 }
38 virtual ~ExtensionSetter() {
39 if (extension_at_call_) {
40 gp_at_call_->resize_extensions();
41 delete gp_at_call_->set_extension(extension_at_call_);
42 }
43 }
44 ExtensionSetter(const ExtensionSetter &other) = delete;
45 ExtensionSetter& operator=(const ExtensionSetter &other) = delete;
46
47 TExtension *operator->() {
48 return &extension_;
49 }
50 iface::Transaction transaction() override {
51 return iface::Transaction(&gp_);
52 }
53 void send_extension(iface::Transaction *transaction) override {
54 TExtension *e = (*transaction)->get_extension<TExtension>();
55 assert(e);
56 if (extension_at_call_)
57 delete extension_at_call_;
58
59 extension_at_call_ = static_cast<TExtension *>(e->clone());
60 }
61 void send_failed(iface::Transaction *transaction) override {}
62
63 private:
64 TPAYLOAD gp_;
65 TPAYLOAD *gp_at_call_;
66 TExtension extension_;
67 TExtension *extension_at_call_;
68};
