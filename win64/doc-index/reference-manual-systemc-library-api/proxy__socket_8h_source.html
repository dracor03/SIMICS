42template <typename TYPES>
43class ProxySocketBase
44 : public Proxy,
45 public ProxySocketInterface,
46 public ScExportConnection,
47 public ScPortConnection,
48 public instrumentation::ToolController,
49 public instrumentation::ToolController::CallbackInterface {
50
51 public:
52 typedef TYPES types;
53
54 explicit ProxySocketBase(simics::ConfObjectRef o)
55 : Proxy(o), ToolController(this), base_socket_(NULL), base_bw_(NULL),
56 base_fw_(NULL), base_bw_typename_(NULL), base_fw_typename_(NULL),
57 multi_initiator_(nullptr), multi_target_(nullptr) {}
58 void init(sc_core::sc_object *obj,
59 SimulationInterface *simulation) override {
60 Proxy::init(obj, simulation);
61 ScExportConnection::init(obj, simulation);
62 ScPortConnection::init(obj, simulation);
63 base_socket_ = dynamic_cast<tlm::tlm_base_socket_if *>(obj);
64 if (base_socket_ == nullptr) {
65 if (obj && simulation) {
66 SIM_LOG_CRITICAL(simulation->simics_object(), Log_Awareness,
67 "SystemC object '%s' does not implement"
68 " tlm::tlm_base_socket_if. Init abort",
69 obj->name());
70 }
71 return;
72 }
73
74 multi_initiator_ =
75 dynamic_cast<tlm_utils::multi_init_base_if<TYPES> *>(obj);
76 if (multi_initiator_) {
77 binders_bw_ = multi_initiator_->get_binders();
78#ifdef SYSTEMC_2_3_3
79 base_bw_ = &base_socket_->get_export_base();
80#elif defined SYSTEMC_2_3_4 || defined SYSTEMC_3_0_0
81 base_bw_ = const_cast<sc_core::sc_export_base *>(
82 &const_cast<const tlm::tlm_base_socket_if *>(
83 base_socket_)->get_base_export());
84#endif
85 base_bw_typename_ =
86 sc_core::simContextProxy::get_typename(base_bw_);
87 for (auto i : binders_bw_) {
88 ScExportConnection::addBinderExport(std::make_pair(i,
89 base_bw_typename_), base_bw_);
90 }
91 }
92
93 multi_target_ =
94 dynamic_cast<tlm_utils::multi_target_base_if<TYPES> *>(obj);
95 if (multi_target_) {
96 binders_fw_ = multi_target_->get_binders();
97#ifdef SYSTEMC_2_3_3
98 base_fw_ = &base_socket_->get_export_base();
99#elif defined SYSTEMC_2_3_4 || defined SYSTEMC_3_0_0
100 base_fw_ = const_cast<sc_core::sc_export_base *>(
101 &const_cast<const tlm::tlm_base_socket_if *>(
102 base_socket_)->get_base_export());
103#endif
104 base_fw_typename_ =
105 sc_core::simContextProxy::get_typename(base_fw_);
106 for (auto i : binders_fw_) {
107 ScExportConnection::keytype key = std::make_pair(i,
108 base_fw_typename_);
109 insertKey(key);
110 ScExportConnection::addBinderExport(key, base_fw_);
111 }
112 }
113 }
114 void allProxiesInitialized() override {
115 for (auto i : binders_bw_) {
116 ScExportConnection *connection =
117 dynamic_cast<ScExportConnection *>(Proxy::findProxy(base_bw_));
118 if (connection) {
119 connection->insertKey(std::make_pair(i,
120 base_bw_typename_));
121 }
122 }
123 }
124 void simulationStarted() override {
125 export_to_proxy();
126 port_to_proxies();
127
128 for (auto b : binders_bw_)
129 addHandler(TlmBaseHandler::mPInitiatorBwSocketHandler(b));
130
131 for (auto f : binders_fw_) {
132 addHandler(TlmBaseHandler::mPTargetBwSocketHandler(f));
133 addHandler(TlmBaseHandler::mPTargetFwSocketHandler(f));
134 }
135
136 if (multi_initiator_) {
137 std::vector<tlm::tlm_fw_transport_if<TYPES> *> &sockets =
138 multi_initiator_->get_sockets();
139 for (auto s : sockets) {
140 addHandler(TlmBaseHandler::mPInitiatorFwSocketHandler(s));
141 addHandler(TlmBaseHandler::mPInitiatorFwMpSocketHandler(s));
142 }
143 }
144
145 if (multi_target_) {
146 std::map<unsigned int, tlm::tlm_bw_transport_if<TYPES> *> &binds =
147 multi_target_->get_multi_binds();
148 for (const auto &b : binds) {
149 addHandler(TlmBaseHandler::mPtargetBwSocketHandler(
150 b.second));
151 }
152 }
153
154 sc_core::sc_object *obj = dynamic_cast<sc_core::sc_object *>(
155 base_socket_);
156 addHandler(TlmBaseHandler::targetBwSocketHandler(obj));
157 addHandler(TlmBaseHandler::targetFwSocketHandler(obj));
158 addHandler(TlmBaseHandler::initiatorBwSocketHandler(obj));
159 addHandler(TlmBaseHandler::initiatorFwSocketHandler(obj));
160 }
161 void simulationEnded() override {
162 }
163 bool isTargetSocket() override {
164 if (base_socket_) {
165 const sc_core::sc_interface *iface = nullptr;
166#ifdef SYSTEMC_2_3_3
167 iface = base_socket_->get_export_base().get_interface();
168#elif defined SYSTEMC_2_3_4 || defined SYSTEMC_3_0_0
169 iface = const_cast<const tlm::tlm_base_socket_if *>(
170 base_socket_)->get_base_export().get_interface();
171#endif
172 return dynamic_cast<const tlm::tlm_fw_transport_if<TYPES> *>(
173 iface) != nullptr;
174 }
175
176 return false;
177 }
178 void tool_controller_init(ToolController *controller) override {
179 for (auto handler : all_handlers_)
180 handler->init(this, controller);
181 }
182 void connection_list_updated(ConnectionListState state) override {
183 for (auto handler : all_handlers_) {
184 if (state == EMPTY)
185 handler->disable();
186 else
187 handler->enable();
188 }
189 }
190 virtual ~ProxySocketBase() {
191 for (auto *binder : binders_bw_) {
192 // the implicit cast triggered a segfault, because the object
193 // pointed to has been freed. Cast via void* to circumvent this,
194 // as all we need in the pair is the pointer address
195 ScExportConnection::keytype key = std::make_pair(
196 static_cast<sc_core::sc_interface *>(
197 static_cast<void*>(binder)), base_bw_typename_);
198 ScExportConnection::removeBinderExport(std::move(key), base_bw_);
199 }
200 }
201
202 protected:
203 void addHandler(TlmHandlerInterface *handler) {
204 if (handler)
205 all_handlers_.push_back(handler);
206 }
207
208 std::string socket_name_;
209 tlm::tlm_base_socket_if *base_socket_;
210 sc_core::sc_export_base *base_bw_;
211 sc_core::sc_export_base *base_fw_;
212 const char *base_bw_typename_;
213 const char *base_fw_typename_;
214 std::vector<tlm_utils::callback_binder_bw<TYPES>* > binders_bw_;
215 std::vector<tlm_utils::callback_binder_fw<TYPES>* > binders_fw_;
216 std::vector<TlmHandlerInterface *> all_handlers_;
217 tlm_utils::multi_init_base_if<TYPES> *multi_initiator_;
218 tlm_utils::multi_target_base_if<TYPES> *multi_target_;
219};

221template <typename TYPES>
222class ProxyInitiatorSocket : public ProxySocketBase<TYPES>,
223 public tlm::tlm_fw_transport_if<TYPES> {
224 public:
225 typedef typename TYPES::tlm_payload_type transaction_type;
226 typedef typename TYPES::tlm_phase_type phase_type;
227 explicit ProxyInitiatorSocket(simics::ConfObjectRef o)
228 : ProxySocketBase<TYPES>(o) {}
229 tlm::tlm_sync_enum nb_transport_fw(transaction_type &trans, // NOLINT
230 phase_type &phase, // NOLINT
231 sc_core::sc_time &t) override { // NOLINT
232 KernelStateModifier m(this->simulation());
233 return fw_if()->nb_transport_fw(trans, phase, t);
234 }
235 void b_transport(transaction_type &trans, // NOLINT
236 sc_core::sc_time &t) override { // NOLINT
237 KernelStateModifier m(this->simulation());
238 invoker_.enqueue(this->base_socket_, &trans, &t);
239 }
240 bool get_direct_mem_ptr(transaction_type &trans, // NOLINT
241 tlm::tlm_dmi &dmi_data) override { // NOLINT
242 KernelStateModifier m(this->simulation());
243 return fw_if()->get_direct_mem_ptr(trans, dmi_data);
244 }
245 unsigned int transport_dbg(transaction_type &trans) override { // NOLINT
246 Kernel(this->simulation()->context());
247 return fw_if()->transport_dbg(trans);
248 }
249 private:
250 tlm::tlm_fw_transport_if<TYPES> *fw_if() {
251#ifdef SYSTEMC_2_3_3
252 tlm::tlm_fw_transport_if<TYPES> *fw_if =
253 dynamic_cast<tlm::tlm_fw_transport_if<TYPES> *>(
254 this->base_socket_->get_port_base().get_interface());
255#elif defined SYSTEMC_2_3_4 || defined SYSTEMC_3_0_0
256 tlm::tlm_fw_transport_if<TYPES> *fw_if =
257 dynamic_cast<tlm::tlm_fw_transport_if<TYPES> *>(
258 const_cast<sc_core::sc_interface *>(
259 const_cast<const tlm::tlm_base_socket_if *>(
260 this->base_socket_)->get_base_port().get_interface()));
261#endif
262 assert(fw_if);
263 return fw_if;
264 }
265 static injection::BTransportInvoker<TYPES> invoker_;
266};
