117 tlm_utils::tlm2_base_protocol_checker &checker(
118 scla::ProxyInterface *proxy) {
119 auto i = checkers_.find(proxy);
120 if (i == checkers_.end()) {
121 i = checkers_.emplace(
122 proxy, new tlm_utils::tlm2_base_protocol_checker(proxy)).first;
123 }
124
125 return *i->second;
126 }
