43template <typename IFACE>
44class MapNameToInterfaceObject : public MapNameToInterface<IFACE> {
45 public:
57 void set_iface(const std::string &name, IFACE *iface) override {
58 if (!iface) {
59 throw std::invalid_argument {
60 "Cannot set with NULL interface"
61 };
62 } else if (name.empty()) {
63 throw std::invalid_argument {
64 "Cannot set with empty name string"
65 };
66 } else {
67 name_to_iface_[hash_str(name)] = iface;
68 }
69 }
70
80 IFACE *get_iface(std::string_view name) const override {
81 return get_iface(hash_str(name));
82 }
83
93 IFACE *get_iface(size_t name_hash) const {
94 auto it = name_to_iface_.find(name_hash);
95 if (it == name_to_iface_.end()) {
96 return nullptr;
97 }
98 return it->second;
99 }
100
109 void erase_iface(const std::string &name) override {
110 name_to_iface_.erase(hash_str(name));
111 }
112
113 private:
115 std::unordered_map<size_t, IFACE *> name_to_iface_;
116};

122/*
123 * A class extends ConfObject to support memory mapped bank registers
124 *
125 * It manages a container which associates a unique name to its corresponding
126 * operational interface. These 3 object types are supported: bank, register
127 * and field.
128 */
129class MappableConfObject : public ConfObject {
130 public:
131 using ConfObject::ConfObject;
132 virtual ~MappableConfObject() = default;
133
139 template <typename IFACE>
140 void set_iface(const std::string &name, IFACE *iface) {
141 if (finalized()) {
142 SIM_LOG_ERROR(obj(), 0,
143 "Cannot set interface for %s when ConfObject"
144 " has been finalized", name.c_str());
145 return;
146 }
147
148 auto current_iface = get_iface<IFACE>(name);
149 if (current_iface && current_iface != iface) {
150 if (iface_maps_write_protected_) {
151 SIM_LOG_INFO(3, obj(), 0,
152 "Interface for %s ignored since iface_map is"
153 " write protected", name.c_str());
154 return;
155 }
156 SIM_LOG_INFO(4, obj(), 0,
157 "Interface for %s overridden", name.c_str());
158 }
159
160 try {
161 std::get<MapNameToInterfaceObject<IFACE>>(
162 iface_maps_).set_iface(name, iface);
163 } catch (const std::exception &e) {
164 SIM_LOG_ERROR(obj(), 0, "%s", e.what());
165 }
166 }
167
174 template <typename IFACE>
175 IFACE *get_iface(std::string_view name) const {
176 return std::get<MapNameToInterfaceObject<IFACE>>(
177 iface_maps_).get_iface(name);
178 }
179
183 RegisterInterface *get_iface(size_t name_hash) const {
184 return std::get<MapNameToInterfaceObject<RegisterInterface>>(
185 iface_maps_).get_iface(name_hash);
186 }
187
192 template <typename IFACE>
193 void erase_iface(const std::string &name) {
194 std::get<MapNameToInterfaceObject<IFACE>>(
195 iface_maps_).erase_iface(name);
196 }
197
198 // Whether to represent the bits in big endian, i.e, whether the bit
199 // number 0 refers to the most significant bit.
200 virtual bool big_endian_bitorder() {
201 return false;
202 }
203
205 bank_memory_t *get_bank_memory(std::string_view name_of_memory) {
206 return &allocated_bank_memories_[name_of_memory.data()];
207 }
208
210 void write_protect_iface_maps(bool write_protect) {
211 iface_maps_write_protected_ = write_protect;
212 }
213
214 private:
216 std::tuple<MapNameToInterfaceObject<BankInterface>,
217 MapNameToInterfaceObject<RegisterInterface>,
218 MapNameToInterfaceObject<FieldInterface>> iface_maps_;
219
220 /*
221 * The allocated_bank_memories_ maps a string key to a
222 * value of type bank_memory_t.
223 *
224 * This structure groups bytes memory for all banks, can be used
225 * to efficiently store and access byte-level data using the name
226 * and byte offset as keys.
227 *
228 * By default, each bank uses its bank name as the key to get the
229 * bank memory. For SharedMemoryBank, multiple banks can share the
230 * same bank memory by using the same key. The key can be any unique
231 * string.
232 */
233 std::unordered_map<std::string,
234 bank_memory_t> allocated_bank_memories_;
235
237 bool iface_maps_write_protected_ {false};
238};
