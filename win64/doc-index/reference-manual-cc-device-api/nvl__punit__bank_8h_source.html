172 // Reset all registers in the bank
173 void reset_all_registers() {
174 unsigned numOfRegs = number_of_registers();
175 for (unsigned i = 0; i < numOfRegs; ++i) {
176 std::pair<size_t, simics::RegisterInterface *> reg_pair = register_at_index(i);
177 reg_pair.second->reset();
178 }
179 }
