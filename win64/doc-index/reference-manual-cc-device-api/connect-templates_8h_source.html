33// A Connect default connects to its descendant
34template<typename FirstIface, typename... RestIfaces>
35class ConnectToDescendant: public Connect<FirstIface, RestIfaces...> {
36 public:
37 ConnectToDescendant(const ConfObject *device,
38 const std::string &descendant)
39 : Connect<FirstIface, RestIfaces...>(device->obj()) {
40 const ConfObjectRef dev_obj = device->obj();
41 // Port must be registered using SIM_register_port
42 if (!SIM_class_port(SIM_object_class(dev_obj), descendant.c_str())) {
43 SIM_LOG_CRITICAL_STR(dev_obj, 0,
44 fmt::format("Port {} is not registered yet",
45 descendant));
46 return;
47 }
48
49 conf_object_t *obj = SIM_object_descendant(dev_obj,
50 descendant.c_str());
51 if (obj) {
52 Connect<FirstIface, RestIfaces...>::set(obj);
53 } else {
54 SIM_LOG_INFO_STR(3, dev_obj, 0,
55 fmt::format("Descendant object {} not found",
56 descendant));
57 }
58 }
59};

61/*
62 * A map target can be viewed as an opaque representation of an
63 * object/interface pair which can function either as an endpoint for a memory
64 * transaction or as an address space where a memory transaction can be
65 * performed.
66 *
67 * The interfaces are tried in the order of on the object:
68 * ram, rom, io_memory, port_space, translator, transaction_translator,
69 * transaction and memory_space
70 */
71class MapTarget {
72 public:
73 explicit MapTarget(const ConfObjectRef &device) : device_(device) {}
74 virtual ~MapTarget() {
75 SIM_free_map_target(map_target_);
76 }
77
78 bool set_map_target(const ConfObjectRef &obj) {
79 SIM_free_map_target(map_target_);
80 if (!obj) {
81 map_target_ = nullptr;
82 return true;
83 }
84
85 map_target_t *tmp = SIM_new_map_target(obj, nullptr, nullptr);
86 if (tmp == nullptr) {
87 std::ignore = SIM_clear_exception();
88 throw detail::SetInterfaceNotFound {SIM_last_error()};
89 }
90 map_target_ = tmp;
91 return true;
92 }
93
94 uint64_t read(uint64_t addr, uint64_t size) {
95 if (size > 8) {
96 SIM_LOG_ERROR(device_, 0, "size must be less than or equal to 8");
97 return 0;
98 }
99 std::vector<uint8_t> val(size);
100 std::vector<atom_t> atoms {
101 ATOM_data(val.data()),
102 ATOM_size(size),
103 ATOM_initiator(device_),
104 ATOM_list_end(0)
105 };
106 transaction_t t {};
107 t.atoms = atoms.data();
108 auto exc = issue(&t, addr);
109 if (exc != Sim_PE_No_Exception) {
110 SIM_LOG_ERROR_STR(device_, 0,
111 fmt::format("unexpected exception: {}",
112 static_cast<int>(exc)));
113 return 0;
114 }
115 return SIM_get_transaction_value_le(&t);
116 }
117
118 void read_bytes(uint64_t addr, uint64_t size, uint8_t *bytes) {
119 std::vector<atom_t> atoms {
120 ATOM_data(bytes),
121 ATOM_size(size),
122 ATOM_initiator(device_),
123 ATOM_list_end(0),
124 };
125 transaction_t t {};
126 t.atoms = atoms.data();
127 issue(&t, addr);
128 }
129
130 void write(uint64_t addr, uint64_t size, uint64_t value) {
131 if (size > 8) {
132 SIM_LOG_ERROR(device_, 0, "size must be less than or equal to 8");
133 return;
134 }
135 std::vector<uint8_t> buf(size);
136 std::vector<atom_t> atoms {
137 ATOM_data(buf.data()),
138 ATOM_size(size),
139 ATOM_flags(Sim_Transaction_Write),
140 ATOM_initiator(device_),
141 ATOM_list_end(0)
142 };
143 transaction_t t {};
144 t.atoms = atoms.data();
145 SIM_set_transaction_value_le(&t, value);
146 issue(&t, addr);
147 }
148
149 void write_bytes(uint64_t addr, uint64_t size, uint8_t *bytes) {
150 std::vector<atom_t> atoms {
151 ATOM_flags(Sim_Transaction_Write),
152 ATOM_data(bytes),
153 ATOM_size(size),
154 ATOM_initiator(device_),
155 ATOM_list_end(0),
156 };
157 transaction_t t {};
158 t.atoms = atoms.data();
159 issue(&t, addr);
160 }
161
162 exception_type_t issue(transaction_t *t, uint64_t addr) {
163 if (t == nullptr) {
164 SIM_LOG_INFO(2, device_, 0,
165 "null transaction is terminated");
166 return Sim_PE_IO_Not_Taken;
167 }
168
169 if (!map_target_) {
170 SIM_LOG_INFO(2, device_, 0,
171 "map_target not set, transaction terminated");
172 return Sim_PE_IO_Not_Taken;
173 }
174
175 exception_type_t exc = SIM_issue_transaction(map_target_, t, addr);
176 std::string op {SIM_transaction_is_read(t) ? "read" : "write"};
177 if (exc == Sim_PE_No_Exception) {
178 SIM_LOG_INFO_STR(4, device_, 0,
179 fmt::format("{} {} bytes @0x{:x} in {}",
180 op, SIM_transaction_size(t),
181 addr, SIM_object_name(device_)));
182 } else {
183 SIM_LOG_INFO_STR(2, device_, 0,
184 fmt::format("failed to {} {} bytes @0x{:x} in {}",
185 op, SIM_transaction_size(t),
186 addr, SIM_object_name(device_)));
187 }
188 return Sim_PE_No_Exception;
189 }
190
191 const map_target_t *map_target() const {
192 return map_target_;
193 }
194
195 private:
196 conf_object_t *device_ {nullptr};
197 map_target_t *map_target_ {nullptr};
198};
