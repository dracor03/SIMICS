30/*
31 * Returns a functor of type set_attr_t serving as a wrapper for a given
32 * pointer to a class member/method.
33 *
34 * The macro uses simics::attr_setter_helper template, which implements
35 * wrapper functions for member.
36 *
37 * Wrapper uses simics::attr_to_std to convert attr_value_t into a standard
38 * C++ type value. If conversion fails, wrapper catches exception and returns
39 * proper value of type set_error_t.
40 */
41#define _S_SINGLE(func_ptr) \
42 simics::detail::attr_setter_helper<decltype(&func_ptr)>::\
43 template f<&func_ptr>
44#define _S_DUAL(cls, m) \
45 simics::detail::attr_setter_helper_dual<decltype(&cls::m), cls>::\
46 template f<&cls::m>
47
48inline set_error_t handle_exception(const std::exception &e) {
49 SIM_attribute_error(e.what());
50 if (dynamic_cast<const SetInterfaceNotFound *>(&e)) {
51 return Sim_Set_Interface_Not_Found;
52 } else if (dynamic_cast<const SetIllegalType *>(&e)) {
53 return Sim_Set_Illegal_Type;
54 } else if (dynamic_cast<const SetNotWritable *>(&e)) {
55 return Sim_Set_Not_Writable;
56 } else {
57 return Sim_Set_Illegal_Value;
58 }
59}

78template <typename O, typename T, typename C>
79struct attr_setter_helper_dual<void (O::*)(T), C> {
80 static_assert(std::is_base_of<O, C>::value, "C must be derived from O");
81 template <void (O::*F)(T)>
82 static set_error_t f(conf_object_t *obj, attr_value_t *val) {
83 O *o = static_cast<O*>(from_obj<C>(obj));
84 try {
85 (o->*F)(attr_to_std<std::remove_cv_t<T>>(*val));
86 } catch (const std::exception &e) {
87 return handle_exception(e);
88 }
89 return Sim_Set_Ok;
90 }
91};

117template <typename O, typename T, typename C>
118struct attr_setter_helper_dual<T O::*, C> {
119 // If T is derived from ConnectBase, use set instead of =
120 template <T O::*R, typename T1 = T> static
121 typename std::enable_if<std::is_base_of<ConnectBase, T1>::value,
122 set_error_t>::type
123 f(conf_object_t *obj, attr_value_t *val) {
124 C *o = from_obj<C>(obj);
125 if ((o->*R).set(attr_to_std<ConfObjectRef>(*val)) == false) {
126 return Sim_Set_Interface_Not_Found;
127 }
128 return Sim_Set_Ok;
129 }
130
131 // If T is a std::array and the value_type is derived from ConnectBase,
132 // assign the new value use set for each member
133 template <T O::*R, typename T1 = T> static
134 typename std::enable_if<is_array_of_connectbase<T1>::value,
135 set_error_t>::type
136 f(conf_object_t *obj, attr_value_t *val) {
137 C *o = from_obj<C>(obj);
138 auto it = (o->*R).begin();
139 for (unsigned i = 0; it != (o->*R).end(); ++i, ++it) {
140 if (it->set(attr_to_std<ConfObjectRef>(
141 SIM_attr_list_item(*val, i))) == false) {
142 return Sim_Set_Interface_Not_Found;
143 }
144 }
145 return Sim_Set_Ok;
146 }
147
148 // If T is a variable length container and the value_type is derived
149 // from ConnectBase, construct a new T
150 template <T O::*R, typename T1 = T> static
151 typename std::enable_if<is_container_of_connectbase<T1>::value,
152 set_error_t>::type
153 f(conf_object_t *obj, attr_value_t *val) {
154 C *o = from_obj<C>(obj);
155 T new_t(SIM_attr_list_size(*val), typename T::value_type(obj));
156 auto it = new_t.begin();
157 for (unsigned i = 0; it != new_t.end(); ++i, ++it) {
158 if (it->set(attr_to_std<ConfObjectRef>(
159 SIM_attr_list_item(*val, i))) == false) {
160 return Sim_Set_Interface_Not_Found;
161 }
162 }
163 o->*R = std::move(new_t);
164 return Sim_Set_Ok;
165 }
166
167 // For all other types
168 template <T O::*R, typename T1 = T> static
169 typename std::enable_if<!is_container_of_connectbase<T1>::value
170 && !is_array_of_connectbase<T1>::value
171 && !std::is_base_of<ConnectBase, T1>::value,
172 set_error_t>::type
173 f(conf_object_t *obj, attr_value_t *val) {
174 C *o = from_obj<C>(obj);
175 try {
176 o->*R = attr_to_std<T>(*val);
177 } catch (const std::exception &e) {
178 return handle_exception(e);
179 }
180 return Sim_Set_Ok;
181 }
182};
