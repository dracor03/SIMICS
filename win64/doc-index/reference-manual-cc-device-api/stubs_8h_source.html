35template <typename ReturnType, typename... Args>
36class Mock {
37 public:
38 using ArgTuple = std::tuple<Args...>;
39
40 // Triggered on every stub invocation
41 ReturnType operator()(Args... args) {
42 call_args_list.push_back(std::make_tuple(args...));
43 size_t current_call = call_count++;
44
45 // 1. Trigger side effects if defined
46 if (side_effect) {
47 return side_effect(args...);
48 }
49
50 // 2. Return sequential values if queued
51 if (current_call < sequence_returns.size()) {
52 return sequence_returns[current_call];
53 }
54
55 auto argument_return = argument_returns.find(
56 std::make_tuple(args...));
57 if (argument_return != argument_returns.end()) {
58 return argument_return->second;
59 }
60
61 // 3. Fallback to default return value
62 return return_value;
63 }
64
65 const ArgTuple & last_call_args() const {
66 return call_args_list.back();
67 }
68
69 void reset() {
70 call_count = 0;
71 call_args_list.clear();
72 sequence_returns.clear();
73 argument_returns.clear();
74 side_effect = nullptr;
75 return_value = default_return_value;
76 }
77
78 // Configuration Settings
79 ReturnType return_value{};
80 ReturnType default_return_value{};
81 std::vector<ReturnType> sequence_returns{};
82 std::map<ArgTuple, ReturnType> argument_returns{};
83 std::function<ReturnType(Args...)> side_effect = nullptr;
84
85 // Call History and Metrics
86 size_t call_count = 0;
87 std::vector<ArgTuple> call_args_list{};
88};

90// Specialization for stubs with no return value, e.g. void SIM_xxx(...)
91template <typename... Args>
92class Mock<void, Args...> {
93 public:
94 using ArgTuple = std::tuple<Args...>;
95
96 void operator()(Args... args) {
97 call_args_list.push_back(std::make_tuple(args...));
98 ++call_count;
99 if (side_effect) {
100 side_effect(args...);
101 }
102 }
103
104 const ArgTuple & last_call_args() const {
105 return call_args_list.back();
106 }
107
108 void reset() {
109 call_count = 0;
110 call_args_list.clear();
111 side_effect = nullptr;
112 }
113
114 // Configuration Settings
115 std::function<void(Args...)> side_effect = nullptr;
116
117 // Call History and Metrics
118 size_t call_count = 0;
119 std::vector<ArgTuple> call_args_list{};
120};
