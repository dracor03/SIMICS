154template <typename... Args>
155struct attr_type_str_helper<std::tuple<Args...>> {
156 static std::string f() {
157 return "[" + type_str() + "]";
158 }
159
160 private:
161 // Function to get the concatenated type string
162 static std::string type_str() {
163 return concat_types<Args...>();
164 }
165
166 // Helper function to concatenate type strings
167 template <typename First, typename Second, typename ...Rest>
168 static std::string concat_types() {
169 return attr_type_str<First>() + concat_types<Second, Rest...>();
170 }
171
172 template <typename Last>
173 static std::string concat_types() {
174 return attr_type_str<Last>();
175 }
176};

212template <typename T>
213inline std::string attr_type_str() {
214 static_assert(!std::is_member_pointer<T>::value,
215 "Use Foo::bar instead of &Foo::bar");
216 static_assert(!std::is_function<T>::value,
217 "Function type is not supported");
218 return detail::attr_type_str_helper<T>::f();
219}
