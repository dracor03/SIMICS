115void scheduled_ctrl_ENABLE_BIT_event() {
116 uint32_t ctrl_data = m_bank->ctrl.get();
117 EXPECT_EQ( ctrl_data, 0x80000000);
118 m_bank->ctrl.ENABLE_BIT.set( 0b0); // clear ctrl ENABLE_BIT
119 uint32_t ctrl_data_2 = m_bank->ctrl.get();
120 EXPECT_EQ( ctrl_data_2, 0x00000000);
121 uint32_t data = m_bank->data_1.read(); // read data from register data_1
122 std::cout << "data_1 data after rd " << std::hex << data << std::endl;
123 EXPECT_EQ( data, 0xdecaf123);
124 uint32_t cleared_data = m_bank->data_1.get(); // check cleared data
125 EXPECT_EQ( cleared_data, 0x00000000);
126 data = (data << 16) | 0xcafe; // performs some logic transform
127 std::cout << "data_1 data after transform " << std::hex << data << std::endl;
128 EXPECT_EQ( data, 0xf123cafe);
